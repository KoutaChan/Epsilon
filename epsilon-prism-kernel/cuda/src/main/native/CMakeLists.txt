cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  message(FATAL_ERROR "Set CMAKE_CUDA_ARCHITECTURES to the CUDA targets to build")
endif()
project(prism_cuda LANGUAGES CXX CUDA)
find_package(JNI REQUIRED)
find_package(CUDAToolkit REQUIRED)

set(PRISM_COMMON "${CMAKE_CURRENT_SOURCE_DIR}/../../../../common")
set(shared "${PRISM_COMMON}/src/main/native")
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(shared_sources "${shared}/cache/candidate_offset_cache.cc")
set(manifest "${shared}/inference/inference_sources.txt")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${manifest}")
file(STRINGS "${manifest}" files)
foreach(file IN LISTS files)
  list(APPEND shared_sources "${shared}/inference/${file}")
endforeach()
set_source_files_properties(${shared_sources} PROPERTIES LANGUAGE CUDA)
set(sources ${shared_sources} collection_jni.cu declaration_jni.cc)
if(WIN32)
  set(resource native/windows-x86_64)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(resource native/linux-x86_64)
else()
  message(FATAL_ERROR "Unsupported CUDA system: ${CMAKE_SYSTEM_NAME}")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|x64)$")
  message(FATAL_ERROR "Expected a 64-bit x86 CUDA toolchain")
endif()

foreach(lib prism_cuda prism_cuda_check)
  if(lib STREQUAL prism_cuda_check)
    set(checked 1)
  else()
    set(checked 0)
  endif()
  add_library(${lib} SHARED ${sources})
  target_include_directories(${lib} PRIVATE . "${shared}" "${shared}/inference"
    "${shared}/collection" ${JNI_INCLUDE_DIRS} ${CUDAToolkit_INCLUDE_DIRS})
  target_compile_definitions(${lib} PRIVATE PRISM_CHECKED=${checked})
  # 積と和をコンパイラーに 1 つの積和の命令へまとめさせない。積和はソースが fusedMultiplyAdd で書く。
  target_compile_options(${lib} PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>"
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>")
  if(WIN32)
    target_compile_options(${lib} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/utf-8>" "$<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8>")
    target_compile_definitions(${lib} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  else()
    target_link_libraries(${lib} PRIVATE ${CMAKE_DL_LIBS})
    target_link_options(${lib} PRIVATE "LINKER:--no-undefined")
  endif()
  install(TARGETS ${lib} LIBRARY DESTINATION "${resource}" COMPONENT jni
    RUNTIME DESTINATION "${resource}" COMPONENT jni ARCHIVE DESTINATION lib COMPONENT development EXCLUDE_FROM_ALL)
endforeach()

include(CTest)
if(BUILD_TESTING)
  set(test_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../test")
  foreach(check buffer_bounds matrix_tile relation_codes)
    add_executable(check_${check} "${test_dir}/native/check_${check}.cu")
    target_include_directories(check_${check} PRIVATE . "${shared}" "${shared}/inference")
  endforeach()
  target_compile_definitions(check_buffer_bounds PRIVATE PRISM_CHECKED=1)
  target_sources(check_relation_codes PRIVATE "${shared}/inference/relation_codes.cu")
  function(gpu_test name target argument)
    add_test(NAME ${name} COMMAND ${target} ${argument})
    set_tests_properties(${name} PROPERTIES LABELS gpu SKIP_RETURN_CODE 77)
  endfunction()
  foreach(mode valid negative past-end)
    gpu_test(buffer_${mode} check_buffer_bounds ${mode})
  endforeach()
  gpu_test(matrix_tile check_matrix_tile 0)
  gpu_test(relation_codes check_relation_codes 0)
endif()
